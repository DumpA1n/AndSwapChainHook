cmake_minimum_required(VERSION 3.22.1)
project(AssembleConsumer LANGUAGES CXX)
set(ASSEMBLE_COMPONENT Assemble CACHE STRING "Public component to consume")
set_property(CACHE ASSEMBLE_COMPONENT PROPERTY STRINGS Assembler Disassembler Assemble)
if(DISASSEMBLER_ONLY)
    set(ASSEMBLE_COMPONENT Disassembler)
endif()
if(NOT ASSEMBLE_COMPONENT MATCHES "^(Assembler|Disassembler|Assemble)$")
    message(FATAL_ERROR "ASSEMBLE_COMPONENT must be Assembler, Disassembler, or Assemble.")
endif()
if(ANDSWAPCHAIN_SOURCE_DIR)
    foreach(component ANDROID GRAPHICS OVERLAY CRASH_RECORDER POINTER_ADAPTER ASSEMBLE ASSEMBLER DISASSEMBLER)
        set(ANDSWAPCHAIN_BUILD_${component} OFF CACHE BOOL "" FORCE)
    endforeach()
    string(TOUPPER "${ASSEMBLE_COMPONENT}" selected_component)
    set(ANDSWAPCHAIN_BUILD_${selected_component} ON CACHE BOOL "" FORCE)
    set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
    add_subdirectory("${ANDSWAPCHAIN_SOURCE_DIR}" producer)
else()
    find_package(AndSwapChainHook 0.2 CONFIG REQUIRED COMPONENTS ${ASSEMBLE_COMPONENT})
endif()
if(ASSEMBLE_COMPONENT STREQUAL "Disassembler")
    if(TARGET Keystone::Keystone OR TARGET AndSwapChainHook::Assembler)
        message(FATAL_ERROR "The disassembler-only package must not load Keystone.")
    endif()
elseif(ASSEMBLE_COMPONENT STREQUAL "Assembler")
    if(TARGET Capstone::Capstone OR TARGET AndSwapChainHook::Disassembler)
        message(FATAL_ERROR "The assembler-only package must not load Capstone.")
    endif()
endif()
string(TOLOWER "${ASSEMBLE_COMPONENT}" fixture)
add_executable(consumer "../${fixture}_roundtrip.cpp")
target_link_libraries(consumer PRIVATE AndSwapChainHook::${ASSEMBLE_COMPONENT})
if(ASSEMBLE_COMPONENT STREQUAL "Assemble")
    set(public_headers Assembler Disassembler)
else()
    set(public_headers ${ASSEMBLE_COMPONENT})
endif()
foreach(header IN LISTS public_headers)
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${header}_header.cpp" "#include <AndSwapChainHook/${header}.h>\n")
    add_library(consumer_header_${header} OBJECT "${CMAKE_CURRENT_BINARY_DIR}/${header}_header.cpp")
    target_link_libraries(consumer_header_${header} PRIVATE AndSwapChainHook::${ASSEMBLE_COMPONENT})
endforeach()
enable_testing()
add_test(NAME assemble_public_consumer COMMAND consumer)
set_tests_properties(assemble_public_consumer PROPERTIES TIMEOUT 30)
