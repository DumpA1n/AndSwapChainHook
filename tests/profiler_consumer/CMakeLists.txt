cmake_minimum_required(VERSION 3.22.1)
project(ProfilerConsumer LANGUAGES CXX)

if(ANDSWAPCHAIN_SOURCE_DIR)
    foreach(component ANDROID GRAPHICS OVERLAY CRASH_RECORDER POINTER_ADAPTER ASSEMBLE ASSEMBLER DISASSEMBLER)
        set(ANDSWAPCHAIN_BUILD_${component} OFF CACHE BOOL "" FORCE)
    endforeach()
    set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
    add_subdirectory("${ANDSWAPCHAIN_SOURCE_DIR}" producer)
else()
    find_package(AndSwapChainHook 0.2 CONFIG REQUIRED COMPONENTS Profiler)
endif()
add_executable(profiler_consumer main.cpp)
target_link_libraries(profiler_consumer PRIVATE AndSwapChainHook::Profiler)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/header.cpp" "#include <AndSwapChainHook/Profiler.h>\n")
add_library(profiler_header OBJECT "${CMAKE_CURRENT_BINARY_DIR}/header.cpp")
target_link_libraries(profiler_header PRIVATE AndSwapChainHook::Profiler)
enable_testing()
add_test(NAME profiler_public_consumer COMMAND profiler_consumer)
set_tests_properties(profiler_public_consumer PROPERTIES TIMEOUT 30)
