cmake_minimum_required(VERSION 3.22.1)
project(AndSwapChainHookConsumer LANGUAGES CXX)
if(ANDSWAPCHAIN_SOURCE_DIR)
    set(ANDSWAPCHAIN_BUILD_ANDROID OFF CACHE BOOL "" FORCE)
    set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
    add_subdirectory("${ANDSWAPCHAIN_SOURCE_DIR}" producer)
else()
    find_package(AndSwapChainHook 0.2 CONFIG REQUIRED COMPONENTS Memory Analysis Instrumentation Input)
endif()
add_executable(consumer main.cpp)
target_link_libraries(consumer PRIVATE AndSwapChainHook::Memory AndSwapChainHook::Analysis AndSwapChainHook::Instrumentation AndSwapChainHook::Input)

if(ANDSWAPCHAIN_CONSUME_ANDROID_MODULES)
    target_compile_definitions(consumer PRIVATE CONSUME_ANDROID_MODULES)
    target_link_libraries(consumer PRIVATE AndSwapChainHook::Graphics AndSwapChainHook::Overlay AndSwapChainHook::CrashRecorder)
endif()
