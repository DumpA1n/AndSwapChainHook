cmake_minimum_required(VERSION 3.22.1)
project(AndSwapChainPointerConsumer LANGUAGES CXX)
if(NOT POINTERHOOK_SOURCE_DIR)
    message(FATAL_ERROR "Set POINTERHOOK_SOURCE_DIR to the validated producer checkout.")
endif()
set(POINTERHOOK_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(POINTERHOOK_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory("${POINTERHOOK_SOURCE_DIR}" pointerhook)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(ANDSWAPCHAIN_BUILD_ANDROID OFF CACHE BOOL "" FORCE)
set(ANDSWAPCHAIN_BUILD_POINTER_ADAPTER ON CACHE BOOL "" FORCE)
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../.." andswapchain)
add_executable(pointer_consumer ../pointer_adapter.cpp)
target_link_libraries(pointer_consumer PRIVATE AndSwapChainHook::HookAdapter)
