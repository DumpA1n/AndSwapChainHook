cmake_minimum_required(VERSION 3.22.1)
project(DobbyConsumer LANGUAGES CXX)
find_package(Dobby 1 CONFIG REQUIRED)
set(ANDSWAPCHAIN_BUILD_ANDROID OFF CACHE BOOL "" FORCE)
set(ANDSWAPCHAIN_BUILD_GRAPHICS ON CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(FETCHCONTENT_FULLY_DISCONNECTED ON CACHE BOOL "" FORCE)
add_subdirectory("${ANDSWAPCHAIN_SOURCE_DIR}" producer)
add_executable(dobby_consumer ../dobby.cpp)
target_link_libraries(dobby_consumer PRIVATE AndSwapChainHook::Instrumentation Dobby::Dobby)
target_link_options(dobby_consumer PRIVATE -Wl,--wrap=mprotect)
target_compile_options(dobby_consumer PRIVATE -mbranch-protection=none)
